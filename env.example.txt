# 抖音下载管理系统 - 环境配置
# 复制此文件为 .env 并修改配置

# 调试模式（生产环境设为 false）
APP_NAME=媒体下载管理系统
DEBUG=false

# 管理端安全配置
# 首次打开网页时会进入初始化流程；请设置一个足够长的随机 Token。
ADMIN_TOKEN=replace_with_a_long_random_token
# 逗号分隔的显式跨域来源；同源使用时留空即可。
CORS_ALLOWED_ORIGINS=

# 统一下载目录：最终路径为 根目录/平台子目录
DOWNLOAD_ROOT=/downloads
DOUYIN_DOWNLOAD_SUBDIR=douyin
X_DOWNLOAD_SUBDIR=X

# 数据库配置
# DB_TYPE 支持: postgresql, mysql
DB_TYPE=postgresql
DB_HOST=localhost
DB_PORT=5432
DB_USER=postgres
DB_PASSWORD=your_db_password
DB_NAME=douyin_service

# 最大同时下载任务数（1～20）
MAX_CONCURRENT_DOWNLOADS=3

# Redis 配置
# 如果 Redis 设置了密码，请填写 REDIS_PASSWORD
REDIS_URL=redis://localhost:6379/0
REDIS_PASSWORD=your_redis_password

# Celery 配置
CELERY_BROKER_URL=redis://localhost:6379/0
CELERY_RESULT_BACKEND=redis://localhost:6379/0

# 下载配置
DOWNLOAD_CHUNK_SIZE=1048576
DOWNLOAD_TIMEOUT=30
DOWNLOAD_RETRY_COUNT=3
DOWNLOAD_RETRY_DELAY=5

# 订阅检查配置
# 默认 6 小时，最小 1 小时，避免频繁拉取作者主页触发限流
DEFAULT_CHECK_INTERVAL=21600
MIN_CHECK_INTERVAL=3600
AUTO_CHECK_ENABLED=true

# 抖音请求节流配置
# REQUEST_DELAY: 单个作者分页拉取作品时，每页之间的等待秒数
# AUTHOR_CHECK_DELAY: 自动检查订阅时，不同作者之间的等待秒数
REQUEST_DELAY=3.0
AUTHOR_CHECK_DELAY=30.0
DOUYIN_RISK_COOLDOWN_SECONDS=300
DOUYIN_RISK_AUTO_RETRY=true

# 卡住任务检测超时（秒），默认600即10分钟
STUCK_TASK_TIMEOUT=600

# 抖音 Cookie（可选，也可通过 Web 界面配置；完整 Cookie 必须包含 UIFID）
# DOUYIN_COOKIE=your_cookie_here

# X/Twitter 配置（X_COOKIE 可通过 Web 界面配置）
X_DOWNLOAD_ENGINE=gallery-dl
X_COOKIE=
X_COOKIE_FILE=
X_TASK_LOG_MAX_LINES=400
X_TASK_LOG_TTL_SECONDS=604800
X_TASK_STATE_TTL_SECONDS=86400
